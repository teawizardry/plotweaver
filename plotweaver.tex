\section{Plotweaver Icon Commands}
These commands produce commonly used icons and are provided by \dispred{plotweaver-icons.sty}. All of the icons are implemented using \disptxt{TikZ}. The icons created by \dispcmd{action}, \dispcmd{doubleAction}, \dispcmd{tripleAction}, \dispcmd{freeAction}, \dispcmd{reaction}, \dispcmd{forever}, and \dispcmd{key} will match the current text color. For example: \textcolor{red}{\action} \textcolor{blue}{\action} \textcolor{green}{\action}. Icons will also respect font size: {\Large\action} \action {\scriptsize\;\action}.

\section{Mistborn Icon Commands}
These commands produce commonly used icons and are provided by \dispred{plotweaver-mistborn.sty}. This package will be automatically loaded if the \disptxt{mistborn} option is passed to the document class. Otherwise, it must be loaded separately. All of the icons are implemented using \disptxt{TikZ} and will match the current text color. For example: \textcolor{red}{\alBronze} \textcolor{blue}{\alBronze} \textcolor{green}{\alBronze}. Icons size can also be alted with an optional argument: \feElectrum[0.5] \feElectrum[0.3] \feElectrum[0.1].

\section{Feature Boxes}
These boxes are implemented using \disptxt{tcolorbox} and are provided by \dispred{plotweaver-boxes.sty}. 

\begin{docKey}[tcb][doc updated = 2026-08-16]{featureBox}{=\meta{text}}{}
  The tcolorbox key containing the styling for feature boxes. See the \disptxt{tcolorbox} documentation for more information. 
  \begin{dispExample*}{sidebyside}
  \begin{tcolorbox}[featureBox, title={Feature Box}]
      \lipsum[1][1-3]
  \end{tcolorbox}
  \end{dispExample*}
\end{docKey}

\begin{docKey}[tcb][doc updated = 2026-08-16]{leftBox}{=\meta{text}}{}
  The tcolorbox key containing the styling for left boxes. See the \disptxt{tcolorbox} documentation for more information. 
  \begin{dispExample*}{sidebyside}
  \begin{tcolorbox}[leftBox, title={Left Box}]
      \lipsum[1][1]
  \end{tcolorbox}
  \end{dispExample*}
\end{docKey}

\begin{docKey}[tcb][doc updated = 2026-08-16]{rightBox}{=\meta{text}}{}

  \begin{dispExample*}{sidebyside}
  \begin{tcolorbox}[rightBox, title={Right Box}]
      \lipsum[1][2]
  \end{tcolorbox}
  \end{dispExample*}
\end{docKey}

\begin{docKey}[tcb][doc new = 2026-08-24]{calloutBox}{=\meta{text}}{}

  \begin{dispExample*}{sidebyside}

      \lipsum[1][3]
  \end{tcolorbox}
  \end{dispExample*}
\end{docKey}

\begin{docKey}[tcb][doc new = 2026-08-26]{altcalloutBox}{=\meta{text}}{}

  \end{tcolorbox}
  \end{dispExample*}
\end{docKey}

\begin{docKey}[tcb][doc new = 2026-08-26]{paperBox}{=\meta{text}}{}

\begin{docEnvironment}[doc updated=2026-08-11]{statBox}{\marg{heading 1}\marg{content 1}\marg{heading 2}\marg{content 2}}
This is a series of custom boxes that will stay the same height regardless of content length. They are always of equal width. You can use each half of \dispred{statBox} separately by using the \dispvar{leftBox} and \dispvar{rightBox} style keys. These follow the same documentation as \dispred{/tcb/featureBox}.
\begin{dispExample*}{sidebyside}
\statBox{Stat One}{
    10
}%
{Stat Two}{
    20
}
\end{dispExample*}
\end{docEnvironment}

\section{Tables}
Tables can be created using \dispred{plotable}. This uses \disptxt{nicematrix} under the hood. See the \disptxt{nicematrix} documentation for more information.

\begin{docEnvironment}[doc new=2026-08-12]{plotable}{\marg{column spec}\oarg{options}}
This is a custom \disptxt{NiceTabular} environment. It simply a wrapper that defines the styling for the table.
\begin{dispExample*}{sidebyside}
  \begin{plotable}{cl}
    Level & Feature \\
    1 & Cool abilities \\
    2 & Even cooler abilities \\
    3 & The coolest abilities \\
  \end{plotable}
\end{dispExample*}
\end{docEnvironment}

\begin{docCommand}[
    doc name        = lowercaption,
    doc new         = 2026-08-13,
    doc parameter   = \marg{caption text},
  ]{}{}
  This command formats a caption to sit below a table. It sets the text to uppercase and uses the \dispvar{csdim} color.
\end{docCommand}

\section{Color Schemes}
\tcbdocmarginnote{\tcbdocnew{2026-08-12}}
This document class currently provides three predefined color schemes: \dispvar{stormlight}, \dispvar{mistborn}, and \dispvar{ariandor}. The default is \dispvar{ariandor}. The color scheme is selected by passing its name into the document class options. Additionally, custom color schemes can be defined by overriding the following color definitions with~\dispcmd{colorlet}:
\begin{itemize}
    \item \dispvar{cspapercolor}: The background color of the page.
    \item \dispvar{cstextcolor}: The color of the text.
    \item \dispvar{cscolor}: The main color used for headings, borders, etc.
    \item \dispvar{csaccent}: The accent color used for underlines, borders, etc.
    \item \dispvar{csdim}: Dimmed color used for notes and other subtle text.
    \item \dispvar{cstblheaderbg}: The background color of table headers.
    \item \dispvar{cstblheadertext}: The text color of table headers.
    \item \dispvar{cstblrowa}: Lighter background color for table rows.
    \item \dispvar{cstblrowb}: Darker background color for table rows.
    \item \dispvar{cscalloutheaderbg}: The background color of callout headers.
    \item \dispvar{cscalloutheadertext}: The text color of callout headers.
    \item \dispvar{cscalloutbg}: The background color of callouts.
\end{itemize}
